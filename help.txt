"""
Broken Encryption 0 — Byte-at-a-time ECB Oracle Attack
=======================================================

Target plaintext:  "agent {input} wants to see {flag}"
                    ^^^^^^         ^^^^^^^^^^^^^^
                    6 bytes         14 bytes
Prefix before flag = 6 + len(input) + 14 = 20 + len(input)
"""

from pwn import remote, context, log
import string

# --- connection parameters ---
HOST = "shell.hackintro.di.uoa.gr"
PORT = 42455
BLOCK = 16

# The service builds: "agent {input} wants to see {flag}"
PREFIX = "agent "
SUFFIX = " wants to see "
FIXED_LEN = len(PREFIX) + len(SUFFIX)

# candidate characters (grouped for readability)
CHARSET = (
    string.ascii_lowercase + string.ascii_uppercase + string.digits
    + "_{}-" + "!@#$%^&*()+=.,;:?/\\'\"|<>[] "
)


def query_server(agent_input: str) -> bytes:
    conn = remote(HOST, PORT, timeout=15)
    conn.recvuntil(b"Agent number: ")
    conn.sendline(agent_input.encode())
    resp = conn.recvline(timeout=10).strip()
    conn.close()
    return bytes.fromhex(resp.decode())


def slice_block(ct: bytes, block_num: int) -> bytes:
    s = block_num * BLOCK
    return ct[s : s + BLOCK]


def compute_L(i: int) -> int:
    """Return L so that flag[i] sits as the final byte of a block."""
    return (BLOCK - ((FIXED_LEN + 1 + i) % BLOCK)) % BLOCK


def block_index_for(i: int) -> int:
    L = compute_L(i)
    return (FIXED_LEN + L + i) // BLOCK


def get_prev15(i: int, recovered: str) -> str:
    L = compute_L(i)
    p = PREFIX + ("A" * L) + SUFFIX + recovered
    pos = FIXED_LEN + L + i
    return p[pos - 15 : pos]


def infer_flag_length() -> int:
    baseline = len(query_server(""))
    for L in range(1, BLOCK + 1):
        if len(query_server("A" * L)) > baseline:
            mod = (-FIXED_LEN - L) % BLOCK
            for cand in range(baseline - BLOCK - FIXED_LEN + 1, baseline - FIXED_LEN + 1):
                if cand % BLOCK == mod:
                    return cand
    raise RuntimeError("could not determine flag length")


def main_attack():
    recovered = ""
    log.info("Probing for flag length...")
    n = infer_flag_length()
    log.info(f"flag length = {n}")

    log.info("Running optimized ECB oracle attack")
    for i in range(n):
        L = compute_L(i)
        ref_ct = query_server("A" * L)
        target_blk = slice_block(ref_ct, block_index_for(i))

        k15 = get_prev15(i, recovered)
        alignment = "A" * (BLOCK - len(PREFIX))
        payload = alignment + "".join(k15 + ch for ch in CHARSET)
        dict_ct = query_server(payload)

        hit = False
        for pos, ch in enumerate(CHARSET):
            if slice_block(dict_ct, 1 + pos) == target_blk:
                recovered += ch
                log.success(f"flag[{i}] = {ch!r} -> {recovered}")
                hit = True
                break

        if not hit:
            log.warning("No match found; assuming flag end or non-printable byte")
            break

        if recovered.endswith("}"):
            break

    print()
    log.success(f"FLAG = {recovered}")
    return recovered


if __name__ == "__main__":
    context.log_level = "info"
    main_attack()

#!/usr/bin/env python3
"""Solve weak RSA CTF challenges from an SSH public key and encrypted blob.

Flow:
1. Parse SSH RSA public key (extract e and n).
2. Fetch factorization of n from FactorDB.
3. Build private exponent d.
4. Decrypt flag.enc as raw RSA: m = c^d mod n.
"""

from __future__ import annotations

import argparse
import base64
import json
import struct
import sys
import urllib.parse
import urllib.request
from typing import Tuple


def parse_ssh_rsa_pub(pub_path: str) -> Tuple[int, int]:
    with open(pub_path, "r", encoding="utf-8") as f:
        line = f.read().strip()

    parts = line.split()
    if len(parts) < 2 or parts[0] != "ssh-rsa":
        raise ValueError("Unsupported public key format. Expected: ssh-rsa ...")

    blob = base64.b64decode(parts[1])
    offset = 0

    def read_u32() -> int:
        nonlocal offset
        if offset + 4 > len(blob):
            raise ValueError("Corrupted SSH key blob")
        value = struct.unpack(">I", blob[offset : offset + 4])[0]
        offset += 4
        return value

    def read_bytes() -> bytes:
        nonlocal offset
        length = read_u32()
        if offset + length > len(blob):
            raise ValueError("Corrupted SSH key blob")
        data = blob[offset : offset + length]
        offset += length
        return data

    key_type = read_bytes().decode("ascii", errors="strict")
    if key_type != "ssh-rsa":
        raise ValueError(f"Unexpected key type in blob: {key_type}")

    e = int.from_bytes(read_bytes(), "big")
    n = int.from_bytes(read_bytes(), "big")
    return e, n


def fetch_factors_from_factordb(n: int) -> Tuple[int, int]:
    query = urllib.parse.urlencode({"query": str(n)})
    url = f"http://www.factordb.com/api?{query}"

    with urllib.request.urlopen(url, timeout=15) as resp:
        data = json.loads(resp.read().decode("utf-8"))

    status = data.get("status")
    factors = data.get("factors", [])

    if status not in {"FF", "P", "Prp"}:
        raise RuntimeError(
            f"FactorDB does not have full factors yet (status={status})."
        )

    expanded = []
    for raw_factor, exp in factors:
        factor_int = int(raw_factor)
        expanded.extend([factor_int] * int(exp))

    if len(expanded) != 2:
        raise RuntimeError(
            f"Expected exactly 2 prime factors for RSA modulus, got {len(expanded)}."
        )

    p, q = expanded
    if p * q != n:
        raise RuntimeError("FactorDB factors do not multiply back to n.")

    return p, q


def mod_inverse(a: int, m: int) -> int:
    try:
        return pow(a, -1, m)
    except ValueError as exc:
        raise RuntimeError("No modular inverse for e modulo phi(n).") from exc


def decrypt_raw_rsa(enc_path: str, n: int, d: int) -> bytes:
    with open(enc_path, "rb") as f:
        c_bytes = f.read()

    c = int.from_bytes(c_bytes, "big")
    if c >= n:
        raise RuntimeError("Ciphertext integer is >= modulus. Not valid raw RSA ciphertext.")

    m = pow(c, d, n)
    k = (n.bit_length() + 7) // 8
    m_bytes = m.to_bytes(k, "big")
    return m_bytes.lstrip(b"\x00")


def main() -> int:
    parser = argparse.ArgumentParser(description="Solve weak RSA challenge from SSH pub key")
    parser.add_argument("--pub", default="id_rsa.pub", help="Path to SSH RSA public key")
    parser.add_argument("--enc", default="flag.enc", help="Path to encrypted flag bytes")
    args = parser.parse_args()

    e, n = parse_ssh_rsa_pub(args.pub)
    print(f"[+] e = {e}")
    print(f"[+] n = {n}")
    print(f"[+] n bit length = {n.bit_length()}")

    p, q = fetch_factors_from_factordb(n)
    print(f"[+] p = {p}")
    print(f"[+] q = {q}")

    phi = (p - 1) * (q - 1)
    d = mod_inverse(e, phi)
    print(f"[+] d = {d}")

    plaintext = decrypt_raw_rsa(args.enc, n, d)
    print(f"[+] raw plaintext bytes = {plaintext!r}")

    try:
        text = plaintext.decode("utf-8")
        print(f"[+] decrypted text = {text}")
    except UnicodeDecodeError:
        print(f"[+] decrypted hex = {plaintext.hex()}")

    return 0


if __name__ == "__main__":
    try:
        raise SystemExit(main())
    except Exception as exc:
        print(f"[!] Error: {exc}", file=sys.stderr)
        raise SystemExit(1)

def read_varuint(data: bytes, pos: int) -> tuple[int, int]:
    result = 0
    shift = 0
    while True:
        b = data[pos]
        pos += 1
        result |= (b & 0x7f) << shift
        if b & 0x80 == 0:
            break
        shift += 7
    return result, pos

def run_vm(prog: list[int], inp: bytes) -> tuple[bool, int]:
    tape = bytearray(256)
    ptr = 0
    ip = 0
    in_idx = 0
    while True:
        op = prog[ip]
        if op == OP_HALT_OK:
            return True, in_idx
        if op == OP_HALT_FAIL:
            return False, in_idx
        if op == OP_SET_CELL_IMM:
            tape[ptr] = prog[ip + 1] & 0xff
            ip += 2
            continue
        if op == OP_SET_PTR_IMM:
            ptr = prog[ip + 1] & 0xff
            ip += 2
            continue
        if op == OP_READ_INPUT:
            tape[ptr] = inp[in_idx]
            in_idx += 1
            ip += 1
            continue
        if op == OP_COPY_CELL:
            src = prog[ip + 1] & 0xff
            tape[ptr] = tape[src]
            ip += 2
            continue
        if op == OP_INC_CELL:
            tape[ptr] = (tape[ptr] + 1) & 0xff
            ip += 1
            continue
        if op == OP_DEC_CELL:
            tape[ptr] = (tape[ptr] - 1) & 0xff
            ip += 1
            continue
        if op == OP_JZ:
            hi = prog[ip + 1] & 0xff
            lo = prog[ip + 2] & 0xff
            target = (hi << 8) | lo
            if tape[ptr] == 0:
                ip = target
            else:
                ip += 3
            continue
        raise RuntimeError(f"unknown op {op} at {ip}")
